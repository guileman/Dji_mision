<script src="https://unpkg.com/leaflet@1.9.4/dist/leaflet.js"></script>

<script src="https://cdn.jsdelivr.net/npm/@turf/turf@6/turf.min.js"></script>

<script>

    // Inicializar mapa (Centrado en una posición inicial)

    const map = L.map('map').setView([-32.959, -60.758], 15);

    L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {

        attribution: '© OpenStreetMap'

    }).addTo(map);

    let coords = [];

    let polygonLayer = L.polygon([], {color: '#3498db', weight: 2}).addTo(map);

    let flightLinesLayer = L.featureGroup().addTo(map);

    // Evento de clic para dibujar

    map.on('click', (e) => {

        coords.push([e.latlng.lng, e.latlng.lat]);

        updatePolygon();

    });

    function updatePolygon() {

        if(coords.length > 0) {

            polygonLayer.setLatLngs(coords.map(c => [c[1], c[0]]));

        }

    }

    function generateFlightPath() {

        if (coords.length < 3) {

            alert("Por favor, marca al menos 3 puntos en el mapa.");

            return;

        }

        flightLinesLayer.clearLayers();

       

        const spacingMeters = document.getElementById('side_dist').value;

        const spacingDegrees = spacingMeters / 111320; // Conversión aproximada a grados

        // Crear polígono Turf

        const poly = turf.polygon([[...coords, coords[0]]]);

        const bbox = turf.bbox(poly);

        // Generar rejilla de líneas

        const grid = turf.lineGrid(bbox, spacingMeters / 1000, {units: 'kilometers'});

       

        grid.features.forEach(line => {

            const intersect = turf.lineSplit(line, poly);

            intersect.features.forEach(segment => {

                // Verificar si el punto medio del segmento está dentro del polígono

                const mid = turf.midpoint(segment.geometry.coordinates[0], segment.geometry.coordinates[1]);

                if (turf.booleanPointInPolygon(mid, poly)) {

                    L.geoJSON(segment, {style: {color: '#f1c40f', weight: 3}}).addTo(flightLinesLayer);

                }

            });

        });

    }

    function exportKML() {

        const layers = flightLinesLayer.getLayers();

        if (layers.length === 0) {

            alert("Primero calcula las rutas de vuelo.");

            return;

        }

        const alt = document.getElementById('altitude').value;

        const dist = document.getElementById('trigger_dist').value;

        const speed = document.getElementById('speed').value;

        let kmlContent = `<?xml version="1.0" encoding="UTF-8"?>

<kml xmlns="http://www.opengis.net/kml/2.2">

  <Document>

    <name>Misión Surveyor DJI</name>

    <description>Generado por Herramienta Estática</description>`;

        layers.forEach((layer, index) => {

            const lineCoords = layer.toGeoJSON().geometry.coordinates;

            const kmlCoords = lineCoords.map(c => `${c[0]},${c[1]},${alt}`).join(' ');

            kmlContent += `

    <Placemark>

      <name>Línea ${index + 1}</name>

      <ExtendedData>

        <Data name="cameraTriggerDistance"><value>${dist}</value></Data>

        <Data name="speed"><value>${speed}</value></Data>

      </ExtendedData>

      <LineString>

        <altitudeMode>relativeToGround</altitudeMode>

        <coordinates>${kmlCoords}</coordinates>

      </LineString>

    </Placemark>`;

        });

        kmlContent += `\n  </Document>\n</kml>`;

        const blob = new Blob([kmlContent], { type: 'application/vnd.google-earth.kml+xml' });

        const url = URL.createObjectURL(blob);

        const a = document.createElement('a');

        a.href = url;

        a.download = 'mision_dji_survey.kml';

        a.click();

    }

    function resetApp() {

        coords = [];

        polygonLayer.setLatLngs([]);

        flightLinesLayer.clearLayers();

    }
